\StudyTitleBlock{10}{Lecture 10: Exam Study Notes}{Spectrum, resolvent, Neumann series, and spectral radius}

\vspace{0.8em}

\begin{focusbox}
This lecture introduces spectral theory in Banach spaces. The exam usually expects the main definitions, the Neumann-series criterion, the openness of the resolvent set, and the core structural properties of the spectrum.
\end{focusbox}

\tableofcontents

\section{Definitions}

\begin{defbox}{Invertible operator}
For $T\in\mathcal L(X)$, the operator $T$ is \textbf{invertible} if there exists $T^{-1}\in\mathcal L(X)$ such that
\[
TT^{-1}=T^{-1}T=I.
\]
\end{defbox}

\begin{defbox}{Resolvent set and spectrum}
For a bounded operator $T\in\mathcal L(X)$ over a complex Banach space,
\[
\rho(T):=\{\lambda\in\C : \lambda I-T \text{ is invertible}\},
\qquad
\sigma(T):=\C\setminus \rho(T).
\]
The \textbf{point spectrum} is the set of eigenvalues.
\end{defbox}

\section{Neumann Series}

\begin{thmbox}{Theorem 14.1: Neumann series}
If $T\in\mathcal L(X)$ satisfies $\norm{T}<1$, then $I-T$ is invertible and
\[
(I-T)^{-1}=\sum_{n=0}^{\infty}T^n.
\]
\end{thmbox}

\paragraph{Why it matters.}
This is the basic perturbative invertibility criterion and appears directly in an oral-exam question.

\section{Resolvent And Spectrum}

\begin{thmbox}{Theorem 14.2: The resolvent set is open}
If $T\in\mathcal L(X)$, then $\rho(T)$ is open, and the resolvent map
\[
\lambda\mapsto R(\lambda,T):=(\lambda I-T)^{-1}
\]
is holomorphic on $\rho(T)$.
\end{thmbox}

\begin{thmbox}{Theorem 14.3: Basic spectral properties}
If $T\in\mathcal L(X)$ and $X\neq\{0\}$ is complex Banach, then:
\begin{itemize}
\item $\sigma(T)$ is nonempty and compact;
\item the spectral radius is
\[
r(T)=\sup\{\abs{\lambda}:\lambda\in\sigma(T)\}=\lim_{m\to\infty}\norm{T^m}^{1/m};
\]
\item one has a Cauchy-integral representation for powers of $T$.
\end{itemize}
\end{thmbox}

\section{Types Of Spectrum}

\begin{exbox}{Vocabulary}
You should know the three standard pieces of the spectrum:
\begin{itemize}
\item \textbf{point spectrum}: eigenvalues;
\item \textbf{continuous spectrum};
\item \textbf{residual spectrum}.
\end{itemize}
For many exam settings, the point is mostly that not every spectral value needs to be an eigenvalue in a general Banach space.
\end{exbox}

\section{What To Memorize Precisely}

\begin{focusbox}
Know precisely:
\begin{enumerate}
\item the definitions of invertibility, resolvent set, spectrum, and point spectrum;
\item Theorems \textbf{14.1}, \textbf{14.2}, \textbf{14.3};
\item why Neumann series gives invertibility of $I-T$ when $\norm{T}<1$;
\item that the spectrum of a bounded operator on a nontrivial complex Banach space is nonempty and compact.
\end{enumerate}
\end{focusbox}
