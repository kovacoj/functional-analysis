\section{Cantor And Baire}

\begin{thmbox}{Theorem 8.1: Cantor intersection theorem}
Let $(X,d)$ be a complete metric space and let
\[
F_1\supset F_2\supset F_3\supset \cdots
\]
be nonempty closed sets such that
\[
\operatorname{diam}(F_n)\to 0.
\]
Then
\[
\bigcap_{n=1}^{\infty}F_n
\]
contains exactly one point.
\end{thmbox}

\paragraph{Proof idea.}
Pick $x_n\in F_n$. Since diameters go to zero, $(x_n)$ is Cauchy. Completeness gives a limit $x$, and closedness plus nesting give $x\in F_n$ for all $n$. Uniqueness follows because two points in all $F_n$ must become arbitrarily close.

\begin{thmbox}{Theorem 8.2: Baire category theorem}
A complete metric space cannot be written as a countable union of nowhere dense sets. Equivalently, a countable intersection of dense open sets is dense.
\end{thmbox}

\begin{thmbox}{Theorem 8.3: Variant of Baire's theorem}
If a complete metric space is a countable union of closed sets, then at least one of those closed sets has nonempty interior.
\end{thmbox}

\section{Meaning And Applications}

\begin{focusbox}
Baire's theorem is powerful because it forces a complete space to be \emph{large} in a topological sense. It prevents complete infinite-dimensional spaces from being exhausted by too many small closed pieces.
\end{focusbox}

\begin{thmbox}{Theorem 8.4: No countable Hamel basis in infinite-dimensional Banach spaces}
An infinite-dimensional Banach space cannot have a countable Hamel basis.
\end{thmbox}

\paragraph{Idea.}
If $\{e_n\}$ were a countable Hamel basis, then the space would be a countable union of finite-dimensional subspaces, each closed and nowhere dense. Baire's theorem forbids this.

\begin{warningbox}{Another standard application}
The space of all polynomials cannot be equipped with a norm making it a Banach space while preserving the natural algebraic structure in the expected way. The reason is the same: it is a countable union of finite-dimensional pieces.
\end{warningbox}

\section{What To Memorize Precisely}

\begin{focusbox}
Know by heart:
\begin{enumerate}
\item the statements of Theorems \textbf{8.1}, \textbf{8.2}, \textbf{8.3}, and \textbf{8.4};
\item what ``nowhere dense'', ``meager'', and ``second category'' mean conceptually;
\item why Baire's theorem applies only in complete metric spaces;
\item the two hallmark applications: no countable Hamel basis, and no Banach structure on all polynomials.
\end{enumerate}
\end{focusbox}
