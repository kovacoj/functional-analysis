\section{Category Terminology}

\begin{defbox}{Nowhere dense, meager, second category}
Let $X$ be a metric space.
\begin{itemize}
\item A set $A\subset X$ is \textbf{nowhere dense} if the interior of $\ol A$ is empty.
\item A set is \textbf{meager} (or of first category) if it is a countable union of nowhere dense sets.
\item A set is of \textbf{second category} if it is not meager.
\end{itemize}
\end{defbox}

\begin{focusbox}
Baire's theorem says that a complete metric space is of second category in itself.
\end{focusbox}

\section{Cantor And Baire}

\begin{thmbox}{Theorem 8.1: Cantor intersection theorem}
\thmCantorIntersection
\end{thmbox}

\begin{prooflevelbox}{FULL}
Construct $x_n\in F_n$; diameters $\to 0$ makes $(x_n)$ Cauchy. Completeness gives a limit $x$. Nesting and closedness give $x\in\bigcap F_n$. Uniqueness: two points in all $F_n$ become arbitrarily close.
\end{prooflevelbox}

\paragraph{Proof idea.}
Pick $x_n\in F_n$. Since diameters go to zero, $(x_n)$ is Cauchy. Completeness gives a limit $x$, and closedness plus nesting give $x\in F_n$ for all $n$. Uniqueness follows because two points in all $F_n$ must become arbitrarily close.

\begin{thmbox}{Theorem 8.2: Baire category theorem}
\thmBaireCategory
\end{thmbox}

\begin{prooflevelbox}{SKETCH}
Know the nested-ball construction: given dense open sets $G_n$, inductively choose closed balls inside $G_n\cap B_{n-1}$ with shrinking radius. Completeness gives a point in the intersection.
\end{prooflevelbox}

\begin{thmbox}{Theorem 8.3: Variant of Baire's theorem}
\thmBaireVariant
\end{thmbox}

\begin{prooflevelbox}{STATEMENT}
Know this as a reformulation of Theorem 8.2: a countable intersection of dense open sets is dense.
\end{prooflevelbox}

\section{Meaning And Applications}

\begin{focusbox}
Baire's theorem is powerful because it forces a complete space to be \emph{large} in a topological sense. It prevents complete infinite-dimensional spaces from being exhausted by too many small closed pieces.
\end{focusbox}

\begin{thmbox}{Theorem 8.4: No countable Hamel basis in infinite-dimensional Banach spaces}
\thmNoCountableHamelBasis
\end{thmbox}

\begin{prooflevelbox}{SKETCH}
If $\{e_n\}$ were a countable Hamel basis, the space would be $\bigcup E_n$ where $E_n=\operatorname{span}\{e_1,\dots,e_n\}$. Each $E_n$ is closed with empty interior (proper subspace of infinite-dimensional space), contradicting Baire.
\end{prooflevelbox}

\paragraph{Idea.}
If $\{e_n\}$ were a countable Hamel basis, then the space would be a countable union of the finite-dimensional subspaces
\[
E_n:=\operatorname{span}\{e_1,\dots,e_n\}.
\]
Each $E_n$ is closed. Since the Banach space is infinite-dimensional, each $E_n$ is a proper subspace, hence has empty interior. Therefore every $E_n$ is nowhere dense, contradicting Baire's theorem.

\begin{warningbox}{Standard corollary}
Any normed space with a countable Hamel basis cannot be Banach unless it is finite-dimensional. In particular, the vector space of all polynomials is countable-dimensional, so no norm can make it into an infinite-dimensional Banach space.
\end{warningbox}

\section{Why This Lecture Matters Later}

\begin{exbox}{Baire as a hidden engine}
The next lecture uses Baire's theorem to prove:
\begin{itemize}
\item the uniform boundedness principle;
\item the open mapping theorem;
\item the bounded inverse theorem;
\item the closed graph theorem.
\end{itemize}
So this lecture is not isolated: it is the topological input behind the main structural theorems of functional analysis.
\end{exbox}
