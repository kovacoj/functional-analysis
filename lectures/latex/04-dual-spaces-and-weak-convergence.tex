\end{focusbox}

\section{Core Definitions}

\begin{defbox}{Dual and second dual}
For a normed space $X$, its dual is
\[
X^*:=\mathcal L(X,\K).
\]
Its second dual is
\[
X^{**}:=(X^*)^*.
\]
\end{defbox}

\begin{defbox}{Canonical embedding}
For $x\in X$, define $Jx\in X^{**}$ by
\[
(Jx)(\phi):=\phi(x)
\qquad \forall \phi\in X^*.
\]
The map $J:X\to X^{**}$ is called the \textbf{canonical embedding}.
\end{defbox}

\begin{defbox}{Reflexive Banach space}
A Banach space $X$ is \textbf{reflexive} if
\[
J(X)=X^{**}.
\]
Equivalently, every element of $X^{**}$ is evaluation at some $x\in X$.
\end{defbox}

\begin{defbox}{Weak and weak-* convergence}
\begin{itemize}
\item $x_n\rightharpoonup x$ in $X$ if $\phi(x_n)\to \phi(x)$ for every $\phi\in X^*$.
\item $\phi_n\overset{*}{\rightharpoonup}\phi$ in $X^*$ if $\phi_n(x)\to \phi(x)$ for every $x\in X$.
\end{itemize}
\end{defbox}

\section{Canonical Embedding And Reflexivity}

\begin{thmbox}{Canonical embedding is an isometry}
For every normed space $X$, the canonical embedding $J:X\to X^{**}$ is linear and isometric:
\[
\norm{Jx}_{X^{**}}=\norm{x}_X.
\]
\end{thmbox}

\begin{prooflevelbox}{FULL}
Reproduce both inequalities: $\norm{Jx}\le\norm{x}$ from $\abs{\phi(x)}\le\norm{\phi}\norm{x}$, and the reverse from the dual characterization $\norm{x}=\sup_{\norm{\phi}\le 1}\abs{\phi(x)}$.
\end{prooflevelbox}

\paragraph{Proof.}
For every $\phi\in X^*$ with $\norm{\phi}\le 1$,
\[
\abs{(Jx)(\phi)}=\abs{\phi(x)}\le \norm{x}.
\]
Hence $\norm{Jx}\le \norm{x}$. The reverse inequality follows from the dual characterization of the norm:
\[
\norm{x}=\sup_{\norm{\phi}\le 1}\abs{\phi(x)}=\norm{Jx}.
\]

\begin{focusbox}
So every normed space embeds isometrically into its second dual; reflexivity asks whether this embedding is onto.
\end{focusbox}

\section{Basic Properties Of Weak And Weak-* Convergence}

\begin{thmbox}{Key observations}
Let $X$ be Banach.
\begin{enumerate}
\item Strong convergence implies weak convergence.
\item Weak limits are unique.
\item In finite-dimensional spaces, weak and strong convergence coincide.
\item In a separable Hilbert space, an orthonormal sequence converges weakly to $0$ but not strongly.
\item If $x_n\rightharpoonup x$, then
\[
\norm{x}\le \liminf_{n\to\infty}\norm{x_n}.
\]
\item If $\phi_n\overset{*}{\rightharpoonup}\phi$, then
\[
\norm{\phi}\le \liminf_{n\to\infty}\norm{\phi_n}.
\]
\item Weakly convergent and weak-* convergent sequences are bounded.
\end{enumerate}
\end{thmbox}

\paragraph{Why these matter.}
These are exactly the properties used later in compactness, weak lower semicontinuity, and existence arguments.

\section{Banach--Alaoglu And Weak Compactness}

\begin{thmbox}{Theorem 7.1: Banach--Alaoglu}
\thmBanachAlaoglu
\end{thmbox}

\begin{prooflevelbox}{SKETCH}
Know the diagonal-subsequence argument in the separable case: extract pointwise convergence on a dense countable set, extend by density and uniform boundedness.
\end{prooflevelbox}

\paragraph{Proof idea in the separable case.}
Choose a countable dense set $(x_k)$ in $X$. From any bounded sequence $(\phi_n)$ extract a diagonal subsequence such that $\phi_n(x_k)$ converges for every $k$. Use density and uniform boundedness to extend the limit to a functional on all of $X$.

\begin{thmbox}{Theorem 7.2: Reflexive spaces have weakly convergent bounded subsequences}
\thmReflexiveWeaklyCompact
\end{thmbox}

\begin{prooflevelbox}{SKETCH}
Reflexivity identifies $X$ with $X^{**}$; then Banach--Alaoglu in $X^{**}$ gives weak-* compactness of the unit ball, which under reflexivity coincides with weak compactness.
\end{prooflevelbox}

\paragraph{Reason.}
Reflexivity identifies $X$ with $X^{**}$, and then weak compactness of the closed unit ball follows from Banach--Alaoglu applied in the second dual.

\begin{thmbox}{Banach--Eberlein--\v Smuljan theorem}
For a Banach space $X$, the following are equivalent:
\begin{enumerate}
\item $X$ is reflexive.
\item The closed unit ball of $X$ is weakly compact.
\item Every bounded sequence in $X$ has a weakly convergent subsequence.
\end{enumerate}
\end{thmbox}

\begin{prooflevelbox}{STATEMENT}
Know the three equivalences and how they connect to Banach--Alaoglu and Theorem 7.2.
\end{prooflevelbox}

\section{Examples Of Reflexive And Non-Reflexive Spaces}

\begin{exbox}{Standard examples}
\begin{itemize}
\item $\ell^p$ and $L^p(\Omega)$ are reflexive for $1<p<\infty$.
\item Hilbert spaces are reflexive.
\item $\ell^1$, $\ell^\infty$, $c_0$, and $L^1$ (in the usual infinite-dimensional cases) are not reflexive.
\end{itemize}
\end{exbox}
